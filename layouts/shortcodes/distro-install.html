# Signier-Schluessel hinterlegen
sudo install -d -m 0755 /etc/apt/keyrings
curl -fsSL https://apt.forgeiec.io/forgeiec.gpg \
  | sudo tee /etc/apt/keyrings/forgeiec.gpg >/dev/null

# Repository-Quelle eintragen
echo "deb [arch=amd64,arm64 signed-by=/etc/apt/keyrings/forgeiec.gpg] \
https://apt.forgeiec.io/trixie trixie main" \
  | sudo tee /etc/apt/sources.list.d/forgeiec.list

sudo apt update
# Signier-Schluessel importieren
sudo rpm --import https://rpm.forgeiec.io/forgeiec-rpm.asc

# Repository eintragen
sudo tee /etc/yum.repos.d/forgeiec.repo >/dev/null <<'EOF'
[forgeiec]
name=ForgeIEC
baseurl=https://rpm.forgeiec.io/el9/$basearch/
enabled=1
gpgcheck=1
repo_gpgcheck=1
gpgkey=https://rpm.forgeiec.io/forgeiec-rpm.asc
EOF

sudo dnf makecache
